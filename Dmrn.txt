# ==========================================
# FUNCIÓN CALCULAR TOTAL
# ==========================================

def calcular_total(costo_paquete, costo_genero, costo_formato, costo_accesorio, costo_extras, costo_servicio, paquete, cantidad_extras):

    subtotal = ( PRECIO_BASE + costo_paquete + costo_genero + costo_formato + costo_accesorio + costo_extras + costo_servicio)

    descuento = 0

    if paquete == "Pack premium" and cantidad_extras >= 3:
        descuento = subtotal * 0.15

    total = subtotal - descuento

    return subtotal, descuento, total


# ==========================================
# FUNCIÓN MOSTRAR RESUMEN
# ==========================================

def mostrar_resumen(paquete,genero,formato,accesorio,extras,servicio,subtotal,descuento,total):

    print("\n===================================")
    print("        RESUMEN DEL PEDIDO")
    print("===================================")

    print(f"📦 Paquete: {paquete}")
    print(f"📚 Género: {genero}")
    print(f"📖 Formato: {formato}")
    print(f"🎁 Accesorio: {accesorio}")



    cantidad_extras = 0

    for extra in extras:
        cantidad_extras += 1

    print(f"\n🛒 Cantidad de extras: {cantidad_extras}")

    if extras != []:

        print("\n✨ Extras seleccionados:")

        for extra in extras:
            print(f"- {extra}")

    else:
        print("No seleccionó extras.")

    print(f"\n🚚 Servicio adicional: {servicio}")

    print("\n-----------------------------------")

    print(f"Subtotal: ${subtotal:,.0f}")
    print(f"Descuento: ${descuento:,.0f}")
    print(f"Total final: ${total:,.0f}")

    print("===================================")
